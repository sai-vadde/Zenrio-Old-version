"use client";

import { motion, useScroll, useTransform } from "framer-motion";
import { useRef } from "react";

const cards = [
  {
    id: "01",
    title: "Define",
    text: "We analyze your goals and craft a clear strategy to guide your brand.",
  },
  {
    id: "02",
    title: "Design",
    text: "Modern visuals and seamless digital experiences.",
  },
  {
    id: "03",
    title: "Build",
    text: "Scalable development using modern tools.",
  },
  {
    id: "04",
    title: "Launch",
    text: "Deployment and optimization for real success.",
  },
];

export default function Services() {
  const ref = useRef(null);

  const { scrollYProgress } = useScroll({
    target: ref,
    offset: ["start 80%", "end 25%"],
  });

  const pathProgress = useTransform(scrollYProgress, [0, 0.75], [0, 1]);

  const card1 = useTransform(scrollYProgress, [0.05, 0.2], [0, 1]);
  const card2 = useTransform(scrollYProgress, [0.2, 0.35], [0, 1]);
  const card3 = useTransform(scrollYProgress, [0.35, 0.5], [0, 1]);
  const card4 = useTransform(scrollYProgress, [0.5, 0.65], [0, 1]);

  return (
    <section
      ref={ref}
      className="bg-gray-100 px-6 md:px-10 py-20 md:py-28 relative overflow-hidden"
    >
      <div className="max-w-6xl mx-auto">
        {/* DESKTOP TIMELINE */}
        <div className="hidden md:block relative h-[700px]">
          {/* Animated SVG Path */}
          <svg
            className="absolute inset-0 w-full h-full pointer-events-none"
            viewBox="0 0 800 700"
            fill="none"
          >
            <motion.path
              d="
                M600 40
                C350 120, 200 180, 250 260
                S550 360, 500 450
                S250 580, 300 650
              "
              stroke="#9CA3AF"
              strokeWidth="2"
              strokeDasharray="12 12"
              strokeLinecap="round"
              style={{
                pathLength: pathProgress,
                opacity: pathProgress,
              }}
            />
          </svg>

          {/* CARD 1 */}
          <motion.div
            style={{
              opacity: card1,
              scale: card1,
              y: useTransform(card1, [0, 1], [80, 0]),
              rotateX: useTransform(card1, [0, 1], [20, 0]),
            }}
            className="absolute top-0 right-10 w-[260px]"
          >
            <Card {...cards[0]} />
          </motion.div>

          {/* CARD 2 */}
          <motion.div
            style={{
              opacity: card2,
              scale: card2,
              y: useTransform(card2, [0, 1], [80, 0]),
              rotateX: useTransform(card2, [0, 1], [20, 0]),
            }}
            className="absolute top-40 left-10 w-[260px]"
          >
            <Card {...cards[1]} />
          </motion.div>

          {/* CARD 3 */}
          <motion.div
            style={{
              opacity: card3,
              scale: card3,
              y: useTransform(card3, [0, 1], [80, 0]),
              rotateX: useTransform(card3, [0, 1], [20, 0]),
            }}
            className="absolute top-[320px] right-10 w-[260px]"
          >
            <Card {...cards[2]} />
          </motion.div>

          {/* CARD 4 */}
          <motion.div
            style={{
              opacity: card4,
              scale: card4,
              y: useTransform(card4, [0, 1], [80, 0]),
              rotateX: useTransform(card4, [0, 1], [20, 0]),
            }}
            className="absolute top-[520px] left-10 w-[260px]"
          >
            <Card {...cards[3]} />
          </motion.div>
        </div>
        {/* MOBILE STACKED TIMELINE */}
        <div className="md:hidden relative border-l-2 border-gray-300 ml-4 space-y-12">
          {cards.map((card, i) => (
            <motion.div
              key={card.id}
              initial={{ opacity: 0, y: 40 }}
              whileInView={{ opacity: 1, y: 0 }}
              transition={{ delay: i * 0.15 }}
              className="relative pl-8"
            >
              <span className="absolute -left-[9px] top-2 w-4 h-4 bg-black rounded-full" />
              <Card {...card} />
            </motion.div>
          ))}
        </div>

        <p className="text-center mt-16 text-gray-600 italic">
          Ready to be delivered!
        </p>
      </div>
    </section>
  );
}

function Card({ id, title, text }: any) {
  return (
    <div className="bg-white/70 backdrop-blur-md border border-white/40 rounded-2xl shadow-lg p-6 hover:shadow-2xl transition duration-300">
      <p className="text-gray-400 text-sm font-medium">{id}</p>
      <h3 className="text-lg font-bold mt-2 text-black">{title}</h3>
      <p className="text-gray-600 mt-2 text-sm leading-relaxed">{text}</p>
    </div>
  );
}
